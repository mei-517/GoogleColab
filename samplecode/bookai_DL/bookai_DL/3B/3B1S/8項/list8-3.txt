import numpy as np

num_bins = 6     # 連続値の分割数

# 連続値の離散化処理を関数化
def digitize_state(observation):
  cart_pos, cart_v, pole_angle, pole_v = observation
  digitized = [
      np.digitize(cart_pos, np.linspace(-2.4, 2.4, num_bins+1)[1:-1]),
      np.digitize(cart_v, np.linspace(-3.0, 3.0, num_bins+1)[1:-1]),
      np.digitize(pole_angle, np.linspace(-0.5, 0.5, num_bins+1)[1:-1]),
      np.digitize(pole_v, np.linspace(-2.0, 2.0, num_bins+1)[1:-1])
  ]
  return sum([x * (num_bins**i) for i, x in enumerate(digitized)])
