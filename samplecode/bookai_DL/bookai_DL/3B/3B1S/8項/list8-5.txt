total_reward = 0.0              # 総報酬を初期化

# ゲーム開始
for i_episode in range(1000):
  obs = env.reset()             # ゲームごとに環境を初期化
  episode_reward = 0.0          # ゲームごとに報酬を初期化

  # 行動開始
  for t in range(200):
    current_state = digitize_state(obs)         # 環境変数の離散化

    if np.random.rand() < 0.1:
      action = env.action_space.sample()        # ランダムに行動 
    else:
      action = np.argmax(q_func[current_state]) # Q値が最大の行動

    obs, reward, done, info = env.step(action)  # 1ステップ実行
    episode_reward += reward                    # 報酬を加算

    # Q値の更新
    next_state = digitize_state(obs)
    q_func[current_state, action] += 0.3 * (reward + 0.99 * np.max(q_func[next_state, :]) - q_func[current_state, action])

    # 行動終了
    if done:
      if (i_episode+1)%100==0:  # 100ゲームごとに結果を表示
        print("Episode {} Steps {} Reward {}".format(i_episode+1, t+1, episode_reward))
      break

  total_reward += episode_reward                # 1ゲームの報酬を加算

print(total_reward, total_reward/1000)          # 総報酬と平均報酬を表示
